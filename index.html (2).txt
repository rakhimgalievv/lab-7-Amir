<script>
    // Данные библиотеки
    const library = {
        about: {
            title: "О библиотеке",
            description:
                "Добро пожаловать в нашу электронную библиотеку!",
            details:
                "Здесь ты можешь познакомиться с интересными книгами, " +
                "узнать об авторах и сохранять понравившиеся произведения " +
                "в избранное.",
            rules: [
                "Просматривай доступные книги.",
                "Добавляй интересные книги в избранное.",
                "Удаляй книги из избранного в любое время."
            ]
        },

        books: [
            {
                id: 1,
                title: "Маленький принц",
                author: "Антуан де Сент-Экзюпери",
                year: 1943,
                genre: "Сказка",
                icon: "🌟"
            },
            {
                id: 2,
                title: "Преступление и наказание",
                author: "Фёдор Достоевский",
                year: 1866,
                genre: "Роман",
                icon: "📖"
            },
            {
                id: 3,
                title: "Мастер и Маргарита",
                author: "Михаил Булгаков",
                year: 1967,
                genre: "Роман",
                icon: "🐈‍⬛"
            },
            {
                id: 4,
                title: "Алиса в Стране чудес",
                author: "Льюис Кэрролл",
                year: 1865,
                genre: "Фэнтези",
                icon: "🐇"
            },
            {
                id: 5,
                title: "Три товарища",
                author: "Эрих Мария Ремарк",
                year: 1936,
                genre: "Роман",
                icon: "📚"
            },
            {
                id: 6,
                title: "Гарри Поттер",
                author: "Дж. К. Роулинг",
                year: 1997,
                genre: "Фэнтези",
                icon: "⚡"
            }
        ]
    };

    // Массив ID избранных книг
    const favorites = new Set();

    const content = document.getElementById("content");
    const tabs = document.querySelectorAll(".tab");
    const favCount = document.getElementById("favCount");

    // Создание карточки книги
    function createBookCard(book, isFavorite) {
        return `
            <article class="book">
                <div class="book-icon">${book.icon}</div>

                <h3>${book.title}</h3>

                <p><b>Автор:</b> ${book.author}</p>
                <p><b>Год:</b> ${book.year}</p>
                <p><b>Жанр:</b> ${book.genre}</p>

                <button
                    class="btn ${isFavorite ? "remove" : ""}"
                    data-action="${isFavorite ? "remove" : "add"}"
                    data-id="${book.id}"
                >
                    ${isFavorite
                        ? "♥ Удалить из избранного"
                        : "♡ В избранное"}
                </button>
            </article>
        `;
    }

    // Отображение каталога
    function showCatalog() {
        content.innerHTML = library.books
            .map(book => createBookCard(
                book,
                favorites.has(book.id)
            ))
            .join("");
    }

    // Отображение избранных книг
    function showFavorites() {
        const books = library.books.filter(
            book => favorites.has(book.id)
        );

        if (books.length === 0) {
            content.innerHTML = `
                <div class="empty">
                    <h2>Пока здесь пусто 💔</h2>
                    <p>
                        Перейди в каталог и добавь любимые книги
                        в избранное.
                    </p>
                </div>
            `;
            return;
        }

        content.innerHTML = books
            .map(book => createBookCard(book, true))
            .join("");
    }

    // Отображение информации о библиотеке
    function showAbout() {
        const data = library.about;

        content.innerHTML = `
            <div class="info">
                <h2>${data.title}</h2>
                <p>${data.description}</p>
                <p>${data.details}</p>
                <h3>Что можно делать?</h3>
                <ul>
                    ${data.rules.map(rule =>
                        `<li>${rule}</li>`
                    ).join("")}
                </ul>
            </div>
        `;
    }

    // Переключение вкладок
    function showTab(tabName) {
        tabs.forEach(tab => {
            tab.classList.toggle(
                "active",
                tab.dataset.tab === tabName
            );
        });

        if (tabName === "catalog") {
            showCatalog();
        } else if (tabName === "favorites") {
            showFavorites();
        } else if (tabName === "about") {
            showAbout();
        }
    }

    // Обновление количества избранных книг
    function updateFavoritesCount() {
        favCount.textContent = favorites.size;
    }

    // Обработка нажатий на вкладки
    tabs.forEach(tab => {
        tab.addEventListener("click", () => {
            showTab(tab.dataset.tab);
        });
    });

    // Добавление и удаление книг через делегирование событий
    content.addEventListener("click", event => {
        const button = event.target.closest("button[data-action]");

        if (!button) return;

        const id = Number(button.dataset.id);

        if (button.dataset.action === "add") {
            favorites.add(id);
        } else if (button.dataset.action === "remove") {
            favorites.delete(id);
        }

        updateFavoritesCount();

        // Перерисовываем текущую вкладку
        const activeTab = document.querySelector(".tab.active");
        showTab(activeTab.dataset.tab);
    });

    // Первоначальная загрузка
    updateFavoritesCount();
    showTab("catalog");
</script>
