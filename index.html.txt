    <script>
        // Данные приложения хранятся в JS-объекте
        const student = {
            about: {
                title: "О себе",
                text: "Привет! Меня зовут Алексей. Я студент, изучаю веб-разработку и программирование.",
                extra: "Мне нравится создавать сайты и изучать новые технологии."
            },

            skills: {
                title: "Мои навыки",
                items: [
                    "HTML — создание структуры сайта",
                    "CSS — оформление и дизайн",
                    "JavaScript — интерактивность",
                    "Git — работа с версиями проекта"
                ]
            },

            contacts: {
                title: "Контакты",
                email: "student@example.com",
                telegram: "@student",
                city: "Алматы"
            }
        };

        const content = document.getElementById("content");
        const tabs = document.querySelectorAll(".tab");

        // Функция отображения содержимого вкладки
        function showTab(tabName) {
            const data = student[tabName];

            if (!data) return;

            if (tabName === "about") {
                content.innerHTML = `
                    <h2>${data.title}</h2>
                    <p>${data.text}</p>
                    <p>${data.extra}</p>
                `;
            }

            if (tabName === "skills") {
                content.innerHTML = `
                    <h2>${data.title}</h2>
                    <ul>
                        ${data.items.map(skill =>
                            `<li>${skill}</li>`
                        ).join("")}
                    </ul>
                `;
            }

            if (tabName === "contacts") {
                content.innerHTML = `
                    <h2>${data.title}</h2>
                    <p>📧 Email: ${data.email}</p>
                    <p>✈️ Telegram: ${data.telegram}</p>
                    <p>📍 Город: ${data.city}</p>
                `;
            }

            // Обновляем активную вкладку
            tabs.forEach(tab => {
                tab.classList.toggle(
                    "active",
                    tab.dataset.tab === tabName
                );
            });
        }

        // Обработка нажатий на вкладки
        tabs.forEach(tab => {
            tab.addEventListener("click", () => {
                showTab(tab.dataset.tab);
            });
        });

        // При запуске открываем вкладку «О себе»
        showTab("about");
    </script>
