<script>
const $ = id => document.getElementById(id);

let students = JSON.parse(localStorage.getItem("journalStudents") || "[]");
let grades = JSON.parse(localStorage.getItem("journalGrades") || "[]");

const save = () => {
  localStorage.setItem("journalStudents", JSON.stringify(students));
  localStorage.setItem("journalGrades", JSON.stringify(grades));
};

const escapeHTML = value => String(value).replace(/[&<>"']/g, ch => ({
  "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;"
}[ch]));

const average = list => list.length
  ? list.reduce((sum, item) => sum + Number(item.value), 0) / list.length
  : null;

const studentGrades = id => grades.filter(g => g.studentId === id);

const averageText = list => {
  const result = average(list);
  return result === null ? "Нет оценок" : result.toFixed(2);
};

function render() {
  $("studentsBody").innerHTML = students.length ? students.map((s, i) => `
    <tr>
      <td>${i + 1}</td>
      <td>${escapeHTML(s.name)}</td>
      <td>${escapeHTML(s.group)}</td>
      <td>${averageText(studentGrades(s.id))}</td>
      <td><button class="delete" data-delete-student="${s.id}">Удалить</button></td>
    </tr>`).join("") :
    '<tr><td colspan="5" class="empty">Студентов пока нет. Добавьте первую запись.</td></tr>';

  $("gradeStudent").innerHTML = '<option value="">Выберите студента</option>' +
    students.map(s => `<option value="${s.id}">${escapeHTML(s.name)} — ${escapeHTML(s.group)}</option>`).join("");

  $("gradesBody").innerHTML = grades.length ? [...grades].reverse().map(g => {
    const s = students.find(s => s.id === g.studentId);
    return `<tr>
      <td>${s ? escapeHTML(s.name) : "Студент удалён"}</td>
      <td>${escapeHTML(g.subject)}</td>
      <td><strong>${g.value}</strong></td>
      <td>${escapeHTML(g.date)}</td>
      <td><button class="delete" data-delete-grade="${g.id}">Удалить</button></td>
    </tr>`;
  }).join("") : '<tr><td colspan="5" class="empty">Оценок пока нет.</td></tr>';

  const allAverage = average(grades);
  $("totalStudents").textContent = students.length;
  $("totalGrades").textContent = grades.length;
  $("overallAverage").textContent = allAverage === null ? "—" : allAverage.toFixed(2);
  $("fives").textContent = grades.filter(g => Number(g.value) === 5).length;

  $("distribution").innerHTML = [5, 4, 3, 2].map(value => {
    const count = grades.filter(g => Number(g.value) === value).length;
    const percent = grades.length ? count / grades.length * 100 : 0;
    return `<p>${value}: ${count} (${percent.toFixed(1)}%)</p>
      <div class="bar"><span style="width:${percent}%"></span></div>`;
  }).join("");

  $("statisticsBody").innerHTML = students.length ? students.map(s => {
    const list = studentGrades(s.id);
    return `<tr><td>${escapeHTML(s.name)}</td>
      <td>${escapeHTML(s.group)}</td><td>${list.length}</td>
      <td>${averageText(list)}</td></tr>`;
  }).join("") : '<tr><td colspan="4" class="empty">Нет данных для статистики.</td></tr>';
}

document.querySelectorAll("[data-tab]").forEach(button => {
  button.addEventListener("click", () => {
    document.querySelectorAll("[data-tab]").forEach(b => b.classList.remove("active"));
    document.querySelectorAll("main > section").forEach(s => s.classList.add("hidden"));
    button.classList.add("active");
    $(button.dataset.tab).classList.remove("hidden");
  });
});

$("studentForm").addEventListener("submit", e => {
  e.preventDefault();
  students.push({
    id: crypto.randomUUID(),
    name: $("studentName").value.trim(),
    group: $("studentGroup").value.trim()
  });
  save();
  e.target.reset();
  render();
});

$("gradeForm").addEventListener("submit", e => {
  e.preventDefault();
  if (!students.some(s => s.id === $("gradeStudent").value)) {
    alert("Сначала добавьте студента.");
    return;
  }
  grades.push({
    id: crypto.randomUUID(),
    studentId: $("gradeStudent").value,
    subject: $("subject").value.trim(),
    value: Number($("gradeValue").value),
    date: $("gradeDate").value
  });
  save();
  $("subject").value = "";
  $("gradeValue").value = "";
  render();
});

document.addEventListener("click", e => {
  const studentId = e.target.dataset.deleteStudent;
  const gradeId = e.target.dataset.deleteGrade;

  if (studentId) {
    if (!confirm("Удалить студента и все его оценки?")) return;
    students = students.filter(s => s.id !== studentId);
    grades = grades.filter(g => g.studentId !== studentId);
    save();
    render();
  }

  if (gradeId) {
    grades = grades.filter(g => g.id !== gradeId);
    save();
    render();
  }
});

$("gradeDate").value = new Date().toLocaleDateString("en-CA");
render();
</script>
